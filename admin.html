Page load hote hi Date Picker mein aaj ki date default set karein document.addEventListener("DOMContentLoaded", () => { const today = new Date().toISOString().split('T')[0]; // YYYY-MM-DD const dateInput = document.getElementById('tcUploadDate'); if (dateInput) dateInput.value = today; }); // Helper: YYYY-MM-DD ko DD-MM-YYYY mein badalne ke liye function formatToDDMMYYYY(isoDate) { if (!isoDate) return null; const parts = isoDate.split('-'); if (parts.length !== 3) return isoDate; return `${parts[2]}-${parts[1]}-${parts[0]}`; } // 2. Tele Consultation Upload Function (CHO / MO) async function uploadTeleConsultationData(file, roleType) { // roleType = 'tele_cho' ya 'tele_mo' const rawDate = document.getElementById('tcUploadDate')?.value; const formattedDate = formatToDDMMYYYY(rawDate); if (!file) { alert("Kripya pehle Excel ya CSV file chunein!"); return; } if (!formattedDate) { alert("Kripya upload ki Date select karein!"); return; } const reader = new FileReader(); reader.onload = async function (e) { try { const data = new Uint8Array(e.target.result); // raw: false se Excel ki dates string format mein hi read hongi const workbook = XLSX.read(data, { type: 'array', raw: false }); const sheetName = workbook.SheetNames[0]; const rawRows = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], { defval: "" }); if (!rawRows || rawRows.length === 0) { alert("File khali hai!"); return; } // Har row ko clean karein taaki koi bhi key 'undefined' na rahe const cleanedRecords = rawRows.map(row => { // Column headers ke extra spaces hatayein const cleanRow = {}; Object.keys(row).forEach(k => { cleanRow[k.trim()] = typeof row[k] === 'string' ? row[k].trim() : row[k]; }); return { block: cleanRow['Block'] || cleanRow['Block Name'] || "Unknown", facility: cleanRow['Facility'] || cleanRow['Facility Name'] || cleanRow['HWC'] || "", staffName: cleanRow['CHO Name'] || cleanRow['MO Name'] || cleanRow['Name'] || "", totalTC: Number(cleanRow['Total Teleconsultation'] || cleanRow['Total TC'] || cleanRow['Count'] || 0), date: formattedDate }; }).filter(item => item.block !== "Unknown" && item.block !== ""); // Firebase se purani availableDates mangwayein const dbRef = firebase.database().ref(`asha_portal_v2/${roleType}`); const snapshot = await dbRef.child('availableDates').once('value'); let existingDates = snapshot.val() || []; // Purani list se undefined/null hatayein aur nayi date add karein if (!Array.isArray(existingDates)) existingDates = Object.values(existingDates); existingDates = existingDates.filter(d => d && d !== "undefined" && d !== "null"); if (!existingDates.includes(formattedDate)) { existingDates.push(formattedDate); } // Dates ko chronological order (purani se nayi) mein sort karein existingDates.sort((a, b) => { const [d1, m1, y1] = a.split('-').map(Number); const [d2, m2, y2] = b.split('-').map(Number); return new Date(y1, m1 - 1, d1) - new Date(y2, m2 - 1, d2); }); // Firebase mein Date-wise data aur updated availableDates save karein await dbRef.child(`recordsByDate/${formattedDate}`).set(cleanedRecords); await dbRef.child('availableDates').set(existingDates); await dbRef.child('latestDate').set(formattedDate); alert(`Safaltapoorvak upload ho gaya! Date: ${formattedDate} (Total Records: ${cleanedRecords.length})`); } catch (err) { console.error("Upload Error:", err); alert("Upload fail hua: " + err.message); } }; reader.readAsArrayBuffer(file); }
